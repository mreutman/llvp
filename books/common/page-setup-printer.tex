\newcommand{\setPageHeader}[1]{%
  \ifnum\pageHeaderEnable>0
    \ifnum\value{page}>\pageHeaderStart%
      \markboth{\formatTextHeader cap. {\footnotesize #1}}%
               {\formatTextHeader cap. {\footnotesize #1}}%
    \else
      \markboth{}{}% Clear headers if we haven't reached the page count yet
    \fi
  \fi
}
